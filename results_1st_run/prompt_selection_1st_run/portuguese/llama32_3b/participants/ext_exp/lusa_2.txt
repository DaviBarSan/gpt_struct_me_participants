import json

def extract_participants(text):
    # Tokenize the text into sentences
    sentences = text.split('. ')

    # Initialize an empty list to store the participants
    participants = []

    # Loop through each sentence
    for sentence in sentences:
        # Tokenize the sentence into words
        words = sentence.split()

        # Loop through each word
        for word in words:
            # Check if the word is a person (e.g. "homem", "desempregado")
            if word in ["homem", "desempregado"]:
                # Add the person to the list of participants
                participants.append(word)

            # Check if the word is a location (e.g. "Porto", "bairro da Pasteleira Nova")
            elif word in ["Porto", "bairro da Pasteleira Nova"]:
                # Add the location to the list of participants
                participants.append(word)

            # Check if the word is an organization (e.g. "PSP", "Comando Metropolitano do Porto")
            elif word in ["PSP", "Comando Metropolitano do Porto"]:
                # Add the organization to the list of participants
                participants.append(word)

            # Check if the word is a type of object (e.g. "arma de fogo", "heroína")
            elif word in ["arma de fogo", "heroína", "cocaína", "tablet", "coluna de som", "euros em dinheiro"]:
                # Add the object to the list of participants
                participants.append(word)

    # Remove duplicates from the list of participants
    participants = list(set(participants))

    # Return the list of participants as a JSON string
    return json.dumps(participants)
