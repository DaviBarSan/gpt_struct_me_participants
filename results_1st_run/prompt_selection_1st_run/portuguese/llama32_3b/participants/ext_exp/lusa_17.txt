import re

def extrair_participantes(texto):
    # Expressões regulares para extrair nomes de pessoas, locais, entidades, etc.
    nomes_pessoas = r'\b([A-Za-z]+ [A-Za-z]+)\b'
    locais = r'\b(Porto|Hong Kong)\b'
    entidades = r'\b(PSP|Comando Metropolitano|South China Morning Post|SCMP|Equipas de Prevenção e Reação Imediata|EPRI)\b'
    objetos = r'\b(arma de fogo|heroína|cocaína|tablet|coluna de som|dinheiro)\b'

    # Compila as expressões regulares
    nomes_pessoas_compilado = re.compile(nomes_pessoas)
    locais_compilado = re.compile(locais)
    entidades_compilado = re.compile(entidades)
    objetos_compilado = re.compile(objetos)

    # Extrai os participantes do texto
    participantes = []
    texto = texto.lower()
    for match in (nomes_pessoas_compilado.finditer(texto), locais_compilado.finditer(texto), entidades_compilado.finditer(texto), objetos_compilado.finditer(texto)):
        participantes.append(match.group())

    # Retira repetições e ordens
    participantes = list(set(participantes))
    participantes.sort()

    return participantes
