```python
import re
import json

def extract_participants(text):
    # Regras para extrair nomes de pessoas e entidades
    rules = [
        r"(\w+ de \w+ anos)",  # Nome de pessoa com idade
        r"(\w+) de (\w+)",  # Nome de pessoa com localização
        r"(\w+) da (\w+)",  # Nome de entidade
        r"(\w+) em (\w+)",  # Nome de entidade com localização
        r"(\w+) da (\w+ Polícia)",  # Nome de entidade policial
        r"(\w+) da (\w+ Judiciária)",  # Nome de entidade judiciária
        r"(\w+) da (\w+ Bombeiros)",  # Nome de entidade de bombeiros
    ]

    # Regras para extrair palavras-chave
    keyword_rules = [
        r"(\w+) de (\w+)",  # Palavra-chave com localização
        r"(\w+) em (\w+)",  # Palavra-chave com localização
        r"(\w+) da (\w+)",  # Palavra-chave com localização
    ]

    # Extração de participantes e palavras-chave
    participants = set()
    keywords = set()
    for rule in rules + keyword_rules:
        matches = re.findall(rule, text, re.IGNORECASE)
        for match in matches:
            if isinstance(match, tuple):
                if len(match) == 2:
                    participants.add(match[0])
                elif len(match) == 3:
                    participants.add(match[0])
                    participants.add(match[2])
            elif isinstance(match, str):
                keywords.add(match)

    # Remoção de palavras-chave redundantes
    keywords = list(keywords)
    for keyword in keywords[:]:
        if any(keyword in p for p in keywords):
            keywords.remove(keyword)

    # Retorno como lista JSON
    return json.dumps(list(participants))

# Exemplo de uso

...
"""

print(extract_participants(text))
```
