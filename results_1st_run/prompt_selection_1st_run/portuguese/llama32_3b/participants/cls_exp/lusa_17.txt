Para classificar e extrair os participantes, utilizarei uma abordagem simples que envolve identificar nomes de pessoas, organizações e outros entes mencionados no texto. Aqui está o código Python para realizar essa tarefa:

```python
import re

# Dicionário de classes
classes = {
    "Person": ["homem", "mulher", "criança", "detido", "pessoa"],
    "Organization": ["PSP", "Comando Metropolitano", "Esquadra de Intervenção e Fiscalização Policial", "Equipas de Prevenção e Reação Imediata", "South China Morning Post", "jornal", "imprensa", "autoridades", "bombeiros"],
    "Object": ["droga", "arma de fogo", "heroína", "cocaína", "tablet", "coluna de som", "munições", "dinheiro", "apartamento", "edifício"],
    "Location": ["Porto", "Pasteleira Nova", "Hong Kong"],
    "Nature": ["incêndio", "crise"],
    "Facility": ["comando", "esquadra", "equipes de prevenção e reação imediata"],
    "Other": ["sete", "cerca de", "uma", "outros", "sequência"]
}

# Função para extrair participantes
def extrair_participantes(texto):
    participantes = []
    # Regras para extrair pessoas
    regra_pessoas = r"(homem|mulher|criança|detido|pessoa)\s*(de\s*20\s*anos|desempregado)"
    # Regras para extrair organizações
    regra_orgs = r"(PSP|Comando Metropolitano|Esquadra de Intervenção e Fiscalização Policial|Equipas de Prevenção e Reação Imediata|South China Morning Post|jornal|imprensa|autoridades|bombeiros)"
    # Regras para extrair objetos
    regra_objetos = r"(droga|arma de fogo|heroína|cocaína|tablet|coluna de som|munições|dinheiro|apartamento|edifício)"
    # Regras para extrair locais
    regra_locais = r"(Porto|Pasteleira Nova|Hong Kong)"
    # Regras para extrair natureza
    regra_natureza = r"(incêndio|crise)"
    # Regras para extrair outras coisas
    regra_outras = r"(sete|cerca de|uma|outros|sequência)"

    # Extrair pessoas
    pessoas = re.findall(regra_pessoas, texto)
    for pessoa in pessoas:
        # Verificar se a pessoa é uma pessoa ou um detido
        if "detido" in pessoa:
            participantes.append([pessoa, "Person"])
        else:
            participantes.append([pessoa, "Person"])
    # Extrair organizações
    orgs = re.findall(regra_orgs, texto)
    for org in orgs:
        participantes.append([org, "Organization"])
    # Extrair objetos
    objetos = re.findall(regra_objetos, texto)
    for objeto in objetos:
        participantes.append([objeto, "Object"])
    # Extrair locais
    locais = re.findall(regra_locais, texto)
    for local in locais:
        participantes.append([local, "Location"])
    # Extrair natureza
    natureza = re.findall(regra_natureza, texto)
    for nature in natureza:
        participantes.append([nature, "Nature"])
    # Extrair outras coisas
    outras = re.findall(regra_outras, texto)
    for outra in outras:
        participantes.append([outra, "Other"])
    return participantes
```
