import re

def extract_and_classify_participants(text):
    participants = []
    classes = ["Person", "Organization", "Object", "Location", "Nature", "Facility", "Other"]

    # Tokenização do texto
    sentences = re.split(r'[.!?]', text)
    for sentence in sentences:
        # Extrair entidades mencionadas no texto
        entities = re.findall(r'\b\w+\b', sentence)
        for entity in entities:
            # Classificar cada entidade
            for cls in classes:
                if cls in entity.lower():
                    participants.append([entity, cls])
                    break

    return participants

text = """

A entrega foi feita pelos representantes legais das empresas Méssene – Investimentos Limitada, Heavy Group – Investimentos Limitada, Universal Sabedoria e LMC.
Estes projetos imobiliários foram financiados com fundos públicos provenientes do Instituto Nacional de Segurança Social, tendo ficado sob a gestão de empresas privadas, sem qualquer contrapartida para o Estado, referiu o comunicado da PGR.
“Assim, com a entrega imediata, definitiva e irrevogável, os bens passam a integrar o património imobiliário do Estado, livre de quaisquer ónus ou encargos”, adiantou a nota.
