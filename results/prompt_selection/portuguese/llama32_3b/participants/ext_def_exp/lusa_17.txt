Para extrair os participantes do texto, você pode usar o seguinte código em Python:

```python
import re
import json

def extrair_participantes(texto):
    # Define uma lista de palavras-chave para identificar os participantes
    palavras_chave = ['homem','mulher', 'criança', 'bombeiro', 'autoridade', 'investigador', 'pessoa','restaurante', 'licença']

    # Inicializa uma lista para armazenar os participantes
    participantes = []

    # Define uma expressão regular para encontrar os participantes
    expressao_regular = r'(\b' + '|'.join(palavras_chave) + r'\b)(?:\s+\w+)?'

    # Encontra todos os participantes no texto
    match = re.findall(expressao_regular, texto, re.IGNORECASE)

    # Adiciona os participantes encontrados à lista
    participantes.extend(match)

    # Retira as palavras-chave repetidas
    participantes = list(set(participantes))

    # Converte a lista para uma lista JSON
    participantes_json = json.dumps(participantes)

    # Retorna a lista JSON
    return participantes_json

"""

participantes = extrair_participantes(texto)
