import re

def extrair_participantes(texto):
    # Expressão regular para encontrar palavras que possam ser participantes
    pattern = r'\b[a-zA-Z0-9\s.,!?]+(?:de|para|com|com a|com os|dos|da|da s|do|dos)\b'

    # Encontrar todas as palavras que atendem ao padrão
    participantes = re.findall(pattern, texto, flags=re.IGNORECASE)

    # Remover palavras que não sejam nomes de pessoas, lugares ou entidades
    participantes = [p for p in participantes if re.match(r'^[a-zA-Z0-9\s]+$', p)]

    return participantes
