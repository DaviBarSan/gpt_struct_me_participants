import re

def extrair_participantes(texto):
    # Define uma lista de palavras-chave para identificar participantes
    palavras_chave = ['pessoa', 'organização', 'local', 'rio', 'provincia','município', 'camponeses', 'administração', 'comunidade']

    # Utiliza regular expressões para encontrar palavras-chave no texto
    participantes = re.findall(r'\b(' + '|'.join(palavras_chave) + r')\b', texto, flags=re.IGNORECASE)

    # Inclui nomes de pessoas e organizações
    pessoas = re.findall(r'([A-Za-z ]+ [A-Za-z ]+)', texto)
    organizacoes = re.findall(r'([A-Z][a-z]+ [A-Z][a-z]+)', texto)
    participantes.append(pessoas)
    participantes.append(organizacoes)

    # Inclui locais e rios
    locais = re.findall(r'([A-Za-z ]+ [A-Z][a-z]+)', texto)
    rios = re.findall(r'([A-Z][a-z]+)', texto)
    participantes.append(locais)
    participantes.append(rios)

    # Retira duplicatas e torna a lista JSON-parseável
    participantes = list(set(participantes))
    participantes = [participante.strip() for participante in participantes]

    return participantes
