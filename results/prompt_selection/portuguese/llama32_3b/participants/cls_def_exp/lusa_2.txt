import json
from nltk.tokenize import word_tokenize
from nltk import pos_tag

# Input text
text = """
PSP detém suspeito de tráfico de droga e apreende arma de fogo no Porto
Um homem de 20 anos suspeito de tráfico de droga foi detido pela PSP no bairro da Pasteleira Nova, no Porto, tendo também sido apreendida heroína, cocaína e uma arma de fogo, informou hoje aquela força policial.
Em comunicado, o Comando Metropolitano do Porto da PSP esclarece que a detenção ocorreu na quarta-feira, no decurso de uma ação policial no âmbito do combate ao crime de tráfico de estupefacientes.
Durante uma busca domiciliária, a PSP apreendeu ao desempregado heroína suficiente para cerca de 599 doses individuais e cocaína suficiente para cerca de 235 doses individuais.
Foi ainda apreendida uma arma de fogo, quatro munições, um ‘tablet', uma coluna de som e 222 euros em dinheiro, descreve aquela força policial.
A operação foi efetuada pelo efetivo da Esquadra de Intervenção e Fiscalização Policial da 2.ª Divisão e das Equipas de Prevenção e Reação Imediata (EPRI), refere a PSP
Segundo a PSP, o detido é hoje presente às Autoridades Judiciárias.
"""

# Tokenize the text
tokens = word_tokenize(text)

# Part-of-speech tagging
tagged_tokens = pos_tag(tokens)

# Define the classes
classes = ['Person', 'Organization', 'Object', 'Location', 'Nature', 'Facility', 'Other']

# Initialize the list of participants
participants = []

# Loop through each tagged token
for token, tag in tagged_tokens:
    # Check if the token is a participant (contains 'de' or 'a')
    if 'de' in token.lower() or 'a' in token.lower():
        # Determine the class of the participant
        if'suspeito' in token.lower() or 'detido' in token.lower() or 'apreendida' in token.lower():
            participant_class = 'Person'
        elif 'PSP' in token.lower():
            participant_class = 'Organization'
        elif 'bairro' in token.lower() or 'Porto' in token.lower():
            participant_class = 'Location'
        elif 'droga' in token.lower() or 'estupefacientes' in token.lower():
            participant_class = 'Object'
        elif 'comunicado' in token.lower() or 'operação' in token.lower():
            participant_class = 'Object'
        else:
            participant_class = 'Other'
        
        # Add the participant to the list
        participants.append([token, participant_class])

# Print the list of participants
print(json.dumps(participants))<|eot_id|>
