import re

def extract_participants(text):
    # Define uma lista de palavras-chave que podem indicar participantes
    participants = ["homem", "mulher", "pessoa", "suspeito", "vítima", "autor", "comandante", "policia", "bombeiro", "judiciário"]

    # Use uma regularização para encontrar todas as palavras que sejam nomes próprios ou descrições de pessoas
    matches = re.findall(r'\b\w+(?: \w+)*\b', text, re.IGNORECASE)

    # Filtrar os resultados para incluir apenas as palavras que sejam nomes próprios ou descrições de pessoas
    filtered_matches = [match for match in matches if any(participant in match.lower() for participant in participants)]

    # Excluir palavras que não sejam nomes próprios ou descrições de pessoas
    participants_list = [match for match in filtered_matches if match.lower() not in participants]

    return participants_list
