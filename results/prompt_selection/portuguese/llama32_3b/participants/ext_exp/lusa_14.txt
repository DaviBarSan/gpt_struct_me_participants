import re

def extrair_participantes(texto):
    # Definir expressões regulares para encontrar as entidades
    participantes = [
        r'(\w+\s+\w+)',  # Nome
        r'(\w+\s+PSP)',  # PSP
        r'(\w+\s+Ministério\w+)',  # Ministério Público
        r'(\w+\s+Grupos\w+)',  # Grupos de trabalhadores
        r'(\w+\s+Cartéis\w+)',  # Cartéis
        r'(\w+\s+Autoridades\w+)',  # Autoridades
        r'(\w+\s+Cartéis\w+)',  # Cartéis
    ]

    # Usar uma lista para armazenar as entidades encontradas
    entidades = []

    # Iterar sobre cada expressão regular
    for expressao in participantes:
        # Usar re.findall para encontrar todas as entidades que atendem à expressão
        entidades.extend(re.findall(expressao, texto))

    # Retornar as entidades encontradas
    return entidades
