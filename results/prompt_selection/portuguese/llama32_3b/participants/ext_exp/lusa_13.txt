import re

def extrair_participantes(texto):
    # Define os nomes das autoridades e instituições
    autoridades = ['PSP', 'Comando Metropolitano do Porto', 'Lusa', 'fiscalização marítima moçambicana', 'César Maphossa', 'Autoridades Judiciárias']
    instituições = ['Porto', 'Moçambique', 'Beira', 'Sofala', 'EPRI', 'Esquadra de Intervenção e Fiscalização Policial', '2.ª Divisão']

    # Inicializa a lista de participantes
    participantes = []

    # Procura por todos os nomes nas autoridades e instituições
    for autoridade in autoridades + instituições:
        # Procura por todas as menções ao nome da autoridade ou instituição
        for palavra in re.findall(r'\b' + autoridade + r'\b', texto, re.IGNORECASE):
            # Adiciona a palavra ao lista de participantes
            participantes.append(palavra)

    # Remove palavras repetidas
    participantes = list(set(participantes))

    # Remove palavras que não são nomes de pessoas ou instituições
    participantes = [participante for participante in participantes if participante.isalpha()]

    # Retorna a lista de participantes
    return participantes
