import re

def extrair_participantes(texto):
    # Regras de busca
    regras = [
        r"(\w+\s\w+)",  # Nome das pessoas
        r"(\w+)",  # Organizações
        r"(\w+)",  # Locais
        r"(\w+)",  # Objetos
        r"(\w+)",  # Eventos
    ]

    # Buscar todos os participantes
    participantes = []
    for regra in regras:
        participantes.extend(re.findall(regra, texto, re.IGNORECASE))

    # Eliminar repetições
    participantes = list(set(participantes))

    return participantes

# Testar a função

"""
