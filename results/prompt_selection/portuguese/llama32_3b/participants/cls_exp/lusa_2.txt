import json
import re

def extract_and_classify_entities(text):
    # Define the classes
    classes = ["Person", "Organization", "Object", "Location", "Nature", "Facility", "Other"]

    # Initialize an empty list to store the extracted entities
    entities = []

    # Split the text into sentences
    sentences = re.split(r'[.!?]', text)

    # Iterate over each sentence
    for sentence in sentences:
        # Split the sentence into words
        words = sentence.split()

        # Iterate over each word
        for word in words:
            # Check if the word is a proper noun (e.g., "PSP", "Porto")
            if word.istitle():
                # Classify the word as an Organization
                entities.append([word, "Organization"])
            # Check if the word is a location (e.g., "Porto", "Pasteleira Nova")
            elif word.lower() in ["porto", "pasteleira nova"]:
                # Classify the word as a Location
                entities.append([word, "Location"])
            # Check if the word is a person (e.g., "homem", "desempregado")
            elif word.lower() in ["homem", "desempregado"]:
                # Classify the word as a Person
                entities.append([word, "Person"])
            # Check if the word is an object (e.g., "droga", "arma de fogo")
            elif word.lower() in ["droga", "arma de fogo", "heroína", "cocaína", "munições", "tablet", "coluna de som", "euros"]:
                # Classify the word as an Object
                entities.append([word, "Object"])
            # Check if the word is a nature (e.g., "tráfico de droga")
            elif word.lower() in ["tráfico de droga", "estupefacientes"]:
                # Classify the word as a Nature
                entities.append([word, "Nature"])
            # Check if the word is a facility (e.g., "Comando Metropolitano do Porto")
            elif word.lower() in ["comando metropolitano do porto"]:
                # Classify the word as a Facility
                entities.append([word, "Facility"])
            # Check if the word is a generic term (e.g., "a força policial")
            elif word.lower() in ["a força policial"]:
                # Classify the word as Other
                entities.append([word, "Other"])

    # Return the extracted entities
    return entities
