def extract_and_classify(text):

    patterns = {

        "Nature": r"([a-zA-Z]+)",
        "Facility": r"([a-zA-Z]+(?: de [a-zA-Z]+)?)",

    }

    # Inicializa a lista com os resultados
    results = []

    # Procura por todos os padrões
    for pattern, class_name in patterns.items():
